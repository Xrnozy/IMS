CREATE DATABASE inventory_management;
USE inventory_management;

CREATE TABLE items (
    item_id INT(11) NOT NULL PRIMARY KEY,
    name TEXT,
    category TEXT,
    quantity INT(11),
    sales_channel TEXT,
    price INT(11)
);

CREATE TABLE orderitems (
    order_id INT(11),
    date TEXT,
    requested_by TEXT,
    sales_channel TEXT,
    item TEXT,
    items INT(11),
    status TEXT,
    last_updated TIMESTAMP NOT NULL DEFAULT current_timestamp()
);
